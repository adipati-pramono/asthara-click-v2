function useInjectStyles() {
  useEffect(() => {
    if (typeof document === "undefined") return;
    if (document.getElementById(ASDS_STYLE_ID)) return;
    const el = document.createElement("style");
    el.id = ASDS_STYLE_ID;
    el.textContent = ASDS_CSS;
    document.head.appendChild(el);
  }, []);
}

/* normalise options: accept strings or {value,label,hint,disabled} */
function normalize(options) {
  return (options || []).map((o) =>
    typeof o === "string" || typeof o === "number"
      ? { value: o, label: String(o) }
      : { value: o.value, label: o.label != null ? o.label : String(o.value), hint: o.hint, disabled: !!o.disabled }
  );
}

export function Select({
  options = [],
  value,
  defaultValue,
  onChange,
  multi = false,
  placeholder = multi ? "Select options…" : "Select an option…",
  label,
  required = false,
  disabled = false,
  size = "md",
  searchable,
  closeOnSelect,
  clearable = true,
  maxTags = 3,
  id,
}) {
  useInjectStyles();
  const opts = useMemo(() => normalize(options), [options]);
  const autoId = useId();
  const rootId = id || autoId;

  const isControlled = value !== undefined;
  const initial = defaultValue !== undefined ? defaultValue : multi ? [] : null;
  const [inner, setInner] = useState(initial);
  const current = isControlled ? value : inner;

  const selected = useMemo(() => {
    if (multi) return Array.isArray(current) ? current : current == null ? [] : [current];
    return current;
  }, [current, multi]);

  const isSel = useCallback(
    (v) => (multi ? selected.includes(v) : selected === v),
    [multi, selected]
  );

  const [open, setOpen] = useState(false);
  const [query, setQuery] = useState("");
  const [activeIdx, setActiveIdx] = useState(-1);
  const rootRef = useRef(null);
  const searchRef = useRef(null);

  const doSearch = searchable === undefined ? opts.length > 7 : searchable;
  const autoClose = closeOnSelect === undefined ? !multi : closeOnSelect;

  const filtered = useMemo(() => {
    if (!doSearch || !query.trim()) return opts;
    const q = query.trim().toLowerCase();
    return opts.filter((o) => o.label.toLowerCase().includes(q));
  }, [opts, query, doSearch]);

  const commit = useCallback(
    (next) => {
      if (!isControlled) setInner(next);
      if (onChange) onChange(next);
    },
    [isControlled, onChange]
  );

  const pick = useCallback(
    (opt) => {
      if (opt.disabled) return;
      if (multi) {
        const set = Array.isArray(selected) ? selected : [];
        const next = set.includes(opt.value) ? set.filter((v) => v !== opt.value) : [...set, opt.value];
        commit(next);
      } else {
        commit(opt.value);
      }
      if (autoClose) { setOpen(false); setQuery(""); }
    },
    [multi, selected, commit, autoClose]
  );

  const removeTag = useCallback(
    (v, e) => {
      if (e) e.stopPropagation();
      commit((Array.isArray(selected) ? selected : []).filter((x) => x !== v));
    },
    [selected, commit]
  );

  const clearAll = useCallback(
    (e) => {
      if (e) e.stopPropagation();
      commit(multi ? [] : null);
    },
    [commit, multi]
  );

  // close on outside click
  useEffect(() => {
    if (!open) return;
    const onDoc = (e) => { if (rootRef.current && !rootRef.current.contains(e.target)) setOpen(false); };
    document.addEventListener("mousedown", onDoc);
    return () => document.removeEventListener("mousedown", onDoc);
  }, [open]);

  // focus search + reset active when opening
  useEffect(() => {
    if (open) {
      setActiveIdx(-1);
      if (doSearch) { const t = setTimeout(() => searchRef.current && searchRef.current.focus(), 20); return () => clearTimeout(t); }
    } else {
      setQuery("");
    }
  }, [open, doSearch]);

  const onKeyDown = useCallback(
    (e) => {
      if (disabled) return;
      if (!open && (e.key === "ArrowDown" || e.key === "Enter" || e.key === " ")) {
        e.preventDefault(); setOpen(true); return;
      }
      if (!open) return;
      if (e.key === "Escape") { e.preventDefault(); setOpen(false); return; }
      if (e.key === "ArrowDown") { e.preventDefault(); setActiveIdx((i) => Math.min(i + 1, filtered.length - 1)); }
      else if (e.key === "ArrowUp") { e.preventDefault(); setActiveIdx((i) => Math.max(i - 1, 0)); }
      else if (e.key === "Enter" && activeIdx >= 0 && filtered[activeIdx]) { e.preventDefault(); pick(filtered[activeIdx]); }
    },
    [disabled, open, filtered, activeIdx, pick]
  );

  const labelFor = useCallback((v) => { const o = opts.find((x) => x.value === v); return o ? o.label : v; }, [opts]);
  const selArr = multi ? (Array.isArray(selected) ? selected : []) : [];
  const hasValue = multi ? selArr.length > 0 : selected != null && selected !== "";

  return (
    <div
      ref={rootRef}
      id={rootId}
      className={[
        "asds-sel",
        `asds-sel--${size}`,
        open ? "asds-sel--open" : "",
        disabled ? "asds-sel--disabled" : "",
        multi ? "asds-sel--multi" : "asds-sel--single",
      ].join(" ").trim()}
    >
      {label && (
        <label className="asds-sel__label" htmlFor={`${rootId}-trigger`}>
          {label}{required && <span className="asds-sel__req">*</span>}
        </label>
      )}

      <button
        type="button"
        id={`${rootId}-trigger`}
        className="asds-sel__trigger"
        aria-haspopup="listbox"
        aria-expanded={open}
        disabled={disabled}
        onClick={() => !disabled && setOpen((o) => !o)}
        onKeyDown={onKeyDown}
      >
        <span className="asds-sel__value">
          {!hasValue && <span className="asds-sel__placeholder">{placeholder}</span>}

          {hasValue && !multi && <span className="asds-sel__single">{labelFor(selected)}</span>}

          {hasValue && multi && selArr.slice(0, maxTags).map((v) => (
            <span className="asds-sel__chip" key={v}>
              <span className="asds-sel__chip-label">{labelFor(v)}</span>
              <span
                className="asds-sel__chip-x"
                role="button"
                aria-label={`Remove ${labelFor(v)}`}
                onClick={(e) => removeTag(v, e)}
              >
                <XMark />
              </span>
            </span>
          ))}
          {hasValue && multi && selArr.length > maxTags && (
            <span className="asds-sel__count">+{selArr.length - maxTags}</span>
          )}
        </span>

        <span className="asds-sel__caret"><Caret /></span>
      </button>

      {open && (
        <div className="asds-sel__pop" role="listbox" aria-multiselectable={multi}>
          {doSearch && (
            <input
              ref={searchRef}
              className="asds-sel__search"
              placeholder="Search…"
              value={query}
              onChange={(e) => { setQuery(e.target.value); setActiveIdx(-1); }}
              onKeyDown={onKeyDown}
            />
          )}

          <ul className="asds-sel__list">
            {filtered.length === 0 && <li className="asds-sel__empty">No matches</li>}
            {filtered.map((o, i) => {
              const sel = isSel(o.value);
              return (
                <li
                  key={o.value}
                  role="option"
                  aria-selected={sel}
                  className={[
                    "asds-sel__opt",
                    sel ? "asds-sel__opt--selected" : "",
                    o.disabled ? "asds-sel__opt--disabled" : "",
                    i === activeIdx ? "asds-sel__opt--active" : "",
                  ].join(" ").trim()}
                  onMouseEnter={() => setActiveIdx(i)}
                  onClick={() => pick(o)}
                >
                  {multi && <span className="asds-sel__box"><Check /></span>}
                  <span className="asds-sel__opt-main">
                    <span className="asds-sel__opt-label">{o.label}</span>
                    {o.hint && <span className="asds-sel__opt-hint">{o.hint}</span>}
                  </span>
                  {!multi && <span className="asds-sel__tick"><Check /></span>}
                </li>
              );
            })}
          </ul>

          {multi && clearable && (
            <div className="asds-sel__foot">
              <span className="asds-sel__foot-info">{selArr.length} selected</span>
              <button type="button" className="asds-sel__foot-clear" onClick={clearAll} disabled={selArr.length === 0}>
                Clear all
              </button>
            </div>
          )}
        </div>
      )}
    </div>
  );
}

  <svg viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeLinecap="round" strokeLinejoin="round"><path d="M18 6 6 18M6 6l12 12" /></svg>
);

const MONTHS = ["January", "February", "March", "April", "May", "June", "July", "August", "September", "October", "November", "December"];
const MONTHS_SHORT = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"];
const DOW_FULL = ["Sun", "Mon", "Tue", "Wed", "Thu", "Fri", "Sat"];

/* ---- date helpers (local, no deps) ---- */
const toDate = (v) => {
  if (v == null || v === "") return null;
  if (v instanceof Date) return isNaN(v) ? null : new Date(v);
  const d = new Date(v);
  return isNaN(d) ? null : d;
};
const sameDay = (a, b) => a && b && a.getFullYear() === b.getFullYear() && a.getMonth() === b.getMonth() && a.getDate() === b.getDate();
const startOfDay = (d) => new Date(d.getFullYear(), d.getMonth(), d.getDate());
const addMonths = (d, n) => new Date(d.getFullYear(), d.getMonth() + n, 1);
const pad2 = (n) => String(n).padStart(2, "0");
const clampDay = (d, min, max) => {
  const t = startOfDay(d).getTime();
  if (min && t < startOfDay(min).getTime()) return true;
  if (max && t > startOfDay(max).getTime()) return true;
  return false;
};

function defaultFormat(d, withTime, hour12) {
  if (!d) return "";
  const datePart = `${MONTHS_SHORT[d.getMonth()]} ${d.getDate()}, ${d.getFullYear()}`;
  if (!withTime) return datePart;
  let h = d.getHours();
  const m = pad2(d.getMinutes());
  if (hour12) {
    const mer = h >= 12 ? "PM" : "AM";
    h = h % 12 || 12;
    return `${datePart} · ${h}:${m} ${mer}`;
  }
  return `${datePart} · ${pad2(h)}:${m}`;
}

export function DatePicker({
  mode = "date",
  value,
  defaultValue,
  onChange,
  placeholder,
  label,
  required = false,
  disabled = false,
  size = "md",
  min,
  max,
  weekStartsOn = 0,
  hour12 = true,
  minuteStep = 5,
  clearable = true,
  align = "left",
  format,
  id,
}) {
  useInjectDpStyles();
  const withTime = mode === "datetime";
  const autoId = useId();
  const rootId = id || autoId;

  const minD = useMemo(() => toDate(min), [min]);
  const maxD = useMemo(() => toDate(max), [max]);

  const isControlled = value !== undefined;
  const [inner, setInner] = useState(() => toDate(defaultValue) || null);
  const selected = isControlled ? toDate(value) : inner;

  const [open, setOpen] = useState(false);
  const [view, setView] = useState(() => startOfDay(selected || toDate(defaultValue) || new Date()));
  const [activeDay, setActiveDay] = useState(null);
  const [panel, setPanel] = useState("days"); // "days" | "months" | "years"
  const rootRef = useRef(null);

  const ph = placeholder || (withTime ? "Select date & time…" : "Select a date…");

  const commit = useCallback(
    (next) => {
      if (!isControlled) setInner(next);
      if (onChange) onChange(next);
    },
    [isControlled, onChange]
  );

  // keep view in sync when opening
  useEffect(() => {
    if (open) {
      setView(startOfDay(selected || new Date()));
      setActiveDay(selected ? startOfDay(selected) : startOfDay(new Date()));
      setPanel("days");
    }
  }, [open]); // eslint-disable-line react-hooks/exhaustive-deps

  // close on outside click
  useEffect(() => {
    if (!open) return;
    const onDoc = (e) => { if (rootRef.current && !rootRef.current.contains(e.target)) setOpen(false); };
    document.addEventListener("mousedown", onDoc);
    return () => document.removeEventListener("mousedown", onDoc);
  }, [open]);

  // build 6-week grid
  const weeks = useMemo(() => {
    const first = new Date(view.getFullYear(), view.getMonth(), 1);
    const offset = (first.getDay() - weekStartsOn + 7) % 7;
    const start = new Date(first);
    start.setDate(first.getDate() - offset);
    const cells = [];
    for (let i = 0; i < 42; i++) {
      const d = new Date(start);
      d.setDate(start.getDate() + i);
      cells.push(d);
    }
    const rows = [];
    for (let i = 0; i < 6; i++) rows.push(cells.slice(i * 7, i * 7 + 7));
    return rows;
  }, [view, weekStartsOn]);

  const dow = useMemo(() => {
    const arr = [];
    for (let i = 0; i < 7; i++) arr.push(DOW_FULL[(i + weekStartsOn) % 7]);
    return arr;
  }, [weekStartsOn]);

  const applyTimeFrom = useCallback(
    (base) => {
      const d = new Date(base);
      if (withTime && selected) { d.setHours(selected.getHours(), selected.getMinutes(), 0, 0); }
      else if (withTime) { const now = new Date(); d.setHours(now.getHours(), now.getMinutes(), 0, 0); }
      return d;
    },
    [withTime, selected]
  );

  const pickDay = useCallback(
    (d) => {
      if (clampDay(d, minD, maxD)) return;
      const next = applyTimeFrom(d);
      commit(next);
      setActiveDay(startOfDay(d));
      if (!withTime) setOpen(false);
    },
    [minD, maxD, applyTimeFrom, commit, withTime]
  );

  const bumpTime = useCallback(
    (part, val) => {
      const base = selected || applyTimeFrom(startOfDay(new Date()));
      const d = new Date(base);
      if (part === "h") d.setHours(val);
      else if (part === "m") d.setMinutes(val);
      else if (part === "mer") {
        const h = d.getHours();
        if (val === "AM" && h >= 12) d.setHours(h - 12);
        if (val === "PM" && h < 12) d.setHours(h + 12);
      }
      d.setSeconds(0, 0);
      commit(d);
    },
    [selected, applyTimeFrom, commit]
  );

  // keyboard nav inside grid
  const onTriggerKey = useCallback(
    (e) => {
      if (disabled) return;
      if (!open && (e.key === "Enter" || e.key === " " || e.key === "ArrowDown")) { e.preventDefault(); setOpen(true); }
    },
    [disabled, open]
  );

  const onGridKey = useCallback(
    (e) => {
      if (!open) return;
      if (e.key === "Escape") { e.preventDefault(); setOpen(false); return; }
      if (panel !== "days") return;
      const base = activeDay || startOfDay(selected || new Date());
      let next = null;
      if (e.key === "ArrowLeft") next = new Date(base.getFullYear(), base.getMonth(), base.getDate() - 1);
      else if (e.key === "ArrowRight") next = new Date(base.getFullYear(), base.getMonth(), base.getDate() + 1);
      else if (e.key === "ArrowUp") next = new Date(base.getFullYear(), base.getMonth(), base.getDate() - 7);
      else if (e.key === "ArrowDown") next = new Date(base.getFullYear(), base.getMonth(), base.getDate() + 7);
      else if (e.key === "Enter" && activeDay) { e.preventDefault(); pickDay(activeDay); return; }
      if (next) {
        e.preventDefault();
        setActiveDay(next);
        if (next.getMonth() !== view.getMonth() || next.getFullYear() !== view.getFullYear()) setView(startOfDay(new Date(next.getFullYear(), next.getMonth(), 1)));
      }
    },
    [open, panel, activeDay, selected, view, pickDay]
  );

  // header chevrons step by month / year / 12-year page depending on the panel
  const stepHeader = useCallback(
    (dir) => {
      setView((v) => {
        if (panel === "days") return addMonths(v, dir);
        if (panel === "months") return new Date(v.getFullYear() + dir, v.getMonth(), 1);
        return new Date(v.getFullYear() + dir * 12, v.getMonth(), 1);
      });
    },
    [panel]
  );

  const monthDisabled = useCallback(
    (y, m) => {
      const start = new Date(y, m, 1);
      const end = new Date(y, m + 1, 0);
      if (minD && end < startOfDay(minD)) return true;
      if (maxD && start > startOfDay(maxD)) return true;
      return false;
    },
    [minD, maxD]
  );

  const yearDisabled = useCallback(
    (y) => {
      if (minD && y < minD.getFullYear()) return true;
      if (maxD && y > maxD.getFullYear()) return true;
      return false;
    },
    [minD, maxD]
  );

  const fmt = format || ((d) => defaultFormat(d, withTime, hour12));
  const today = new Date();
  const yearPageStart = view.getFullYear() - (view.getFullYear() % 12);
  const stepLabel = panel === "days" ? "month" : panel === "months" ? "year" : "years";
  const titleText =
    panel === "days"
      ? `${MONTHS[view.getMonth()]} ${view.getFullYear()}`
      : panel === "months"
      ? String(view.getFullYear())
      : `${yearPageStart} \u2013 ${yearPageStart + 11}`;

  // time field display values
  let dispH = "", dispM = "", mer = "AM";
  if (withTime) {
    const t = selected || new Date();
    mer = t.getHours() >= 12 ? "PM" : "AM";
    dispH = hour12 ? (t.getHours() % 12 || 12) : t.getHours();
    dispM = t.getMinutes();
  }

  return (
    <div
      ref={rootRef}
      id={rootId}
      className={[
        "asds-dp",
        `asds-dp--${size}`,
        open ? "asds-dp--open" : "",
        disabled ? "asds-dp--disabled" : "",
        align === "right" ? "asds-dp--align-right" : "",
        withTime ? "asds-dp--datetime" : "asds-dp--date",
      ].join(" ").trim()}
    >
      {label && (
        <label className="asds-dp__label" htmlFor={`${rootId}-trigger`}>
          {label}{required && <span className="asds-dp__req">*</span>}
        </label>
      )}

      <button
        type="button"
        id={`${rootId}-trigger`}
        className="asds-dp__trigger"
        aria-haspopup="dialog"
        aria-expanded={open}
        disabled={disabled}
        onClick={() => !disabled && setOpen((o) => !o)}
        onKeyDown={onTriggerKey}
      >
        <span className="asds-dp__cal-ic">{withTime ? <IconClock /> : <IconCal />}</span>
        <span className="asds-dp__value">
          {selected ? <time dateTime={selected.toISOString()}>{fmt(selected)}</time> : <span className="asds-dp__placeholder">{ph}</span>}
        </span>
        {clearable && selected && !disabled && (
          <span
            className="asds-dp__clear"
            role="button"
            aria-label="Clear date"
            onClick={(e) => { e.stopPropagation(); commit(null); }}
          >
            <IconX />
          </span>
        )}
      </button>

      {open && (
        <div className="asds-dp__pop" role="dialog" aria-label={label || "Choose date"} onKeyDown={onGridKey}>
          <div className="asds-dp__head">
            <button type="button" className="asds-dp__nav" aria-label={`Previous ${stepLabel}`} onClick={() => stepHeader(-1)}><IconChevL /></button>
            <button type="button" className="asds-dp__title" aria-label="Switch calendar view" onClick={() => setPanel((p) => (p === "days" ? "months" : "years"))}>{titleText}</button>
            <button type="button" className="asds-dp__nav" aria-label={`Next ${stepLabel}`} onClick={() => stepHeader(1)}><IconChevR /></button>
          </div>

          {panel === "days" && (
          <div className="asds-dp__dow">
            {dow.map((d, i) => <span key={i}>{d}</span>)}
          </div>
          )}

          {panel === "days" && (
          <div className="asds-dp__grid" role="grid">
            {weeks.map((row, ri) =>
              row.map((d, ci) => {
                const out = d.getMonth() !== view.getMonth();
                const isSel = sameDay(d, selected);
                const isToday = sameDay(d, today);
                const isActive = sameDay(d, activeDay);
                const isDisabled = clampDay(d, minD, maxD);
                return (
                  <button
                    type="button"
                    key={`${ri}-${ci}`}
                    role="gridcell"
                    aria-selected={isSel}
                    aria-current={isToday ? "date" : undefined}
                    disabled={isDisabled}
                    tabIndex={isActive ? 0 : -1}
                    className={[
                      "asds-dp__day",
                      out ? "asds-dp__day--out" : "",
                      isToday ? "asds-dp__day--today" : "",
                      isActive && !isSel ? "asds-dp__day--active" : "",
                      isSel ? "asds-dp__day--sel" : "",
                      isDisabled ? "asds-dp__day--disabled" : "",
                    ].join(" ").trim()}
                    onClick={() => pickDay(d)}
                    onMouseEnter={() => setActiveDay(startOfDay(d))}
                  >
                    {d.getDate()}
                  </button>
                );
              })
            )}
          </div>
          )}

          {panel === "months" && (
            <div className="asds-dp__cells asds-dp__cells--month">
              {MONTHS_SHORT.map((m, i) => {
                const isSel = selected && selected.getFullYear() === view.getFullYear() && selected.getMonth() === i;
                const isCur = today.getFullYear() === view.getFullYear() && today.getMonth() === i;
                const dis = monthDisabled(view.getFullYear(), i);
                return (
                  <button
                    type="button"
                    key={m}
                    disabled={dis}
                    className={["asds-dp__cell", isCur ? "asds-dp__cell--cur" : "", isSel ? "asds-dp__cell--sel" : "", dis ? "asds-dp__cell--disabled" : ""].join(" ").trim()}
                    onClick={() => { setView(new Date(view.getFullYear(), i, 1)); setPanel("days"); }}
                  >
                    {m}
                  </button>
                );
              })}
            </div>
          )}

          {panel === "years" && (
            <div className="asds-dp__cells asds-dp__cells--year">
              {Array.from({ length: 12 }, (_, k) => yearPageStart + k).map((y) => {
                const isSel = selected && selected.getFullYear() === y;
                const isCur = today.getFullYear() === y;
                const dis = yearDisabled(y);
                return (
                  <button
                    type="button"
                    key={y}
                    disabled={dis}
                    className={["asds-dp__cell", isCur ? "asds-dp__cell--cur" : "", isSel ? "asds-dp__cell--sel" : "", dis ? "asds-dp__cell--disabled" : ""].join(" ").trim()}
                    onClick={() => { setView(new Date(y, view.getMonth(), 1)); setPanel("months"); }}
                  >
                    {y}
                  </button>
                );
              })}
            </div>
          )}

          {panel === "days" && withTime && (
            <div className="asds-dp__time">
              <span className="asds-dp__time-ic"><IconClock /></span>
              <div className="asds-dp__time-fields">
                <input
                  className="asds-dp__spin"
                  type="number"
                  aria-label="Hour"
                  min={hour12 ? 1 : 0}
                  max={hour12 ? 12 : 23}
                  value={pad2(dispH)}
                  onChange={(e) => {
                    let n = parseInt(e.target.value, 10);
                    if (isNaN(n)) return;
                    if (hour12) {
                      n = Math.max(1, Math.min(12, n));
                      const h24 = mer === "PM" ? (n % 12) + 12 : n % 12;
                      bumpTime("h", h24);
                    } else {
                      n = Math.max(0, Math.min(23, n));
                      bumpTime("h", n);
                    }
                  }}
                />
                <span className="asds-dp__colon">:</span>
                <input
                  className="asds-dp__spin"
                  type="number"
                  aria-label="Minute"
                  min={0}
                  max={59}
                  step={minuteStep}
                  value={pad2(dispM)}
                  onChange={(e) => {
                    let n = parseInt(e.target.value, 10);
                    if (isNaN(n)) return;
                    bumpTime("m", Math.max(0, Math.min(59, n)));
                  }}
                />
                {hour12 && (
                  <div className="asds-dp__mer" role="group" aria-label="AM or PM">
                    <button type="button" aria-pressed={mer === "AM"} onClick={() => bumpTime("mer", "AM")}>AM</button>
                    <button type="button" aria-pressed={mer === "PM"} onClick={() => bumpTime("mer", "PM")}>PM</button>
                  </div>
                )}
              </div>
            </div>
          )}

          {panel === "days" && (
          <div className="asds-dp__foot">
            <button
              type="button"
              className="asds-dp__today-btn"
              onClick={() => { const base = applyTimeFrom(startOfDay(new Date())); if (!clampDay(base, minD, maxD)) { commit(base); setView(startOfDay(base)); setActiveDay(startOfDay(base)); if (!withTime) setOpen(false); } }}
            >
              {withTime ? "Now" : "Today"}
            </button>
            {withTime && (
              <button type="button" className="asds-dp__done-btn" onClick={() => setOpen(false)}>Done</button>
            )}
          </div>
          )}
        </div>
      )}
    </div>
  );
}

var AS_AVATAR_TINTS = ["gold", "teal", "info", "rose", "success", "purple"];

function asAvatarTint(name) {
  var h = 0;
  for (var i = 0; i < name.length; i++) h = (h * 31 + name.charCodeAt(i)) >>> 0;
  return "as-avatar--" + AS_AVATAR_TINTS[h % AS_AVATAR_TINTS.length];
}

function asAvatarInitials(name) {
  return name.split(" ").slice(0, 2).map(function (w) { return w[0] || ""; }).join("").toUpperCase();
}

/* Asthara · Sidebar behaviour.
   - A section header opens and closes its section.
   - A clicked link becomes the current page (aria-current="page"); Soon items stay inactive.
   - Example links (href="#") stay on the page instead of jumping to the top.
   Behaviour only; in your app, set aria-current from the route and aria-expanded from your state. */
document.addEventListener("click", function (e) {
  var head = e.target.closest(".as-sidebar__section-head");
  if (head) {
    head.setAttribute("aria-expanded", head.getAttribute("aria-expanded") === "true" ? "false" : "true");
    return;
  }
  var link = e.target.closest(".as-sidebar__link, .as-sidebar__sublink");
  if (!link || link.getAttribute("aria-disabled") === "true") return;
  if (link.getAttribute("href") === "#") e.preventDefault();
  var sidebar = link.closest(".as-sidebar");
  sidebar.querySelectorAll('[aria-current="page"]').forEach(function (el) { el.removeAttribute("aria-current"); });
  link.setAttribute("aria-current", "page");
});
